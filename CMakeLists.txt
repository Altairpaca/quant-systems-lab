cmake_minimum_required(VERSION 3.16)
project(quant_systems_lab LANGUAGES CXX)
find_package(Threads REQUIRED)
option(QSL_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(QSL_TSAN "Thread sanitizer (separate build from ASan)" OFF)
if(QSL_SANITIZE AND QSL_TSAN)
  message(FATAL_ERROR "Use separate sanitizer builds")
endif()
add_executable(qsl cpp/src/main.cpp)
target_include_directories(qsl PRIVATE cpp/include)
target_compile_features(qsl PRIVATE cxx_std_20)
target_link_libraries(qsl PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(qsl PRIVATE -Wall -Wextra -Wpedantic)
  if(QSL_SANITIZE)
    target_compile_options(qsl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(qsl PRIVATE -fsanitize=address,undefined)
  endif()
  if(QSL_TSAN)
    target_compile_options(qsl PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(qsl PRIVATE -fsanitize=thread)
  endif()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME replay_and_queue COMMAND qsl test)
add_test(NAME bench_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/test_bench_cli.py --binary $<TARGET_FILE:qsl>)
set_tests_properties(replay_and_queue PROPERTIES TIMEOUT 60)
set_tests_properties(bench_cli PROPERTIES TIMEOUT 120)
